"""Explicit live smoke tests; opt in after installing the real runtime.
RUN_SANDBOX_TESTS=1 python -m unittest discover -s tests -p test_sandbox_live.py -v
These smoke tests are not a security certification.
"""
import os
import shlex
import socket
import tempfile
import unittest
from pathlib import Path
from mini_agent import build_tools


@unittest.skipUnless(os.environ.get('RUN_SANDBOX_TESTS') == '1', 'Real sandbox tests require explicit opt-in and installed dependencies')
class RealSandboxTest(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory(prefix='srt-live-')
        self.addCleanup(self.tmp.cleanup)
        self.root = Path(self.tmp.name) / 'workspace'
        self.root.mkdir()
        self.tools = build_tools(self.root)

    def bash(self, command, timeout=10000):
        result = self.tools.execute('Bash', {'command': command, 'timeout': timeout})
        self.assertNotIn('Sandbox unavailable', result.content, result.content)
        self.assertNotIn('Sandbox error:', result.content, result.content)
        return result

    def test_success(self):
        r = self.bash("printf 'hello sandbox'")
        self.assertFalse(r.is_error, r.content)
        self.assertIn('hello sandbox', r.content)

    def test_write_outside_workspace_denied(self):
        target = Path(self.tmp.name) / 'outside.txt'
        r = self.bash('printf x > ' + shlex.quote(str(target)))
        self.assertTrue(r.is_error, r.content)
        self.assertFalse(target.exists())

    def test_credential_file_read_denied(self):
        (self.root / '.env').write_text('TEST_SECRET=must-not-leak')
        r = self.bash('cat .env')
        self.assertTrue(r.is_error, r.content)
        self.assertNotIn('must-not-leak', r.content)

    def test_direct_network_to_host_denied(self):
        with socket.socket() as listener:
            listener.bind(('127.0.0.1', 0))
            listener.listen()
            port = listener.getsockname()[1]
            r = self.bash(f'exec 3<>/dev/tcp/127.0.0.1/{port}', timeout=1500)
            self.assertTrue(r.is_error, r.content)

    def test_timeout(self):
        r = self.bash('sleep 10', timeout=200)
        self.assertTrue(r.is_error, r.content)
        self.assertIn('Timed out: true', r.content)


if __name__ == '__main__':
    unittest.main()
