/** One-command sandbox lifecycle. This is an adapter, NOT a sandbox reimplementation. */
import { spawn } from 'node:child_process';

let manager;
let child;
let killTimer;
let interrupted = false;

function killGroup(signal = 'SIGTERM') {
  if (!child?.pid) return;
  try { process.kill(-child.pid, signal); }
  catch (error) { if (error.code !== 'ESRCH') throw error; }
}
function stopCommand() {
  killGroup('SIGTERM');
  clearTimeout(killTimer);
  killTimer = setTimeout(() => killGroup('SIGKILL'), 500);
}
for (const signal of ['SIGTERM', 'SIGINT']) {
  process.on(signal, () => {
    interrupted = true;
    stopCommand();
    if (!child) process.exitCode = 1;
  });
}
function collector(maxBytes = 16000) {
  const chunks = [];
  let kept = 0, seen = 0;
  return {
    add(chunk) {
      seen += chunk.length;
      const piece = chunk.subarray(0, Math.max(0, maxBytes - kept));
      if (piece.length) { chunks.push(piece); kept += piece.length; }
    },
    text() { return Buffer.concat(chunks).toString('utf8'); },
    truncated() { return seen > kept; },
  };
}
async function main() {
  let payload = '';
  for await (const chunk of process.stdin) {
    payload += chunk;
    if (Buffer.byteLength(payload) > 1000000) throw new Error('Bridge request too large');
  }
  const req = JSON.parse(payload);
  if (!['linux', 'darwin'].includes(process.platform)) throw new Error('Unsupported bridge platform');
  if (typeof req.command !== 'string' || !req.command.trim() || req.command.includes('\0')) throw new Error('Invalid command');
  if (!Number.isInteger(req.timeout) || req.timeout < 100 || req.timeout > 60000) throw new Error('Invalid timeout');
  process.chdir(req.workspace);

  // Missing package / unsupported runtime = failure. Never run req.command directly.
  const { SandboxManager } = await import('@anthropic-ai/sandbox-runtime');
  manager = SandboxManager;
  for (const method of ['initialize', 'wrapWithSandbox', 'cleanupAfterCommand', 'reset']) {
    if (typeof manager?.[method] !== 'function') throw new Error(`Runtime API missing: ${method}`);
  }
  await manager.initialize(req.config);
  if (interrupted) throw new Error('Interrupted before execution');
  const wrapped = await manager.wrapWithSandbox(req.command, '/bin/bash');
  if (!wrapped || wrapped === req.command) throw new Error('Runtime did not produce a sandbox wrapper');
  if (interrupted) throw new Error('Interrupted before execution');
  const out = collector(), err = collector();
  let timedOut = false;
  let timer;
  try {
    const result = await new Promise((resolve, reject) => {
      child = spawn('/bin/bash', ['--noprofile', '--norc', '-c', wrapped], {
        cwd: req.workspace, env: process.env, detached: true,
        stdio: ['ignore', 'pipe', 'pipe'],
      });
      child.stdout.on('data', chunk => out.add(chunk));
      child.stderr.on('data', chunk => err.add(chunk));
      child.once('error', reject);
      child.once('close', (code, signal) => resolve({code, signal}));
      timer = setTimeout(() => { timedOut = true; stopCommand(); }, req.timeout);
    });
    return {
      ok: true, stdout: out.text(), stderr: err.text(),
      exit_code: result.code, signal: result.signal,
      timed_out: timedOut || interrupted,
      output_truncated: out.truncated() || err.truncated(),
    };
  } finally {
    clearTimeout(timer);
    clearTimeout(killTimer);
    killGroup('SIGKILL'); // No intentionally persistent/background processes in this MVP.
    // Must run after the wrapped command; some profiles own temporary descriptors.
    manager.cleanupAfterCommand();
  }
}
let response;
try { response = await main(); }
catch (error) { response = {ok: false, error: String(error.message ?? error).slice(0, 2000)}; }
finally {
  if (manager) {
    try { await manager.reset(); }
    catch (error) { response = {ok: false, error: `Sandbox cleanup failed: ${error.message}`}; }
  }
}
process.stdout.write(JSON.stringify(response) + '\n');
process.exitCode = response.ok ? 0 : 1;
