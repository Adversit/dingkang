from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path
from typing import Callable
import json


@dataclass
class Message:
    role: str
    content: str
    kind: str = "text"
    tool_use_id: str | None = None


@dataclass
class SessionMemory:
    current_task: str = ""
    key_files: list[str] = field(default_factory=list)
    errors_and_fixes: list[str] = field(default_factory=list)
    key_results: list[str] = field(default_factory=list)
    next_step: str = ""

    def render(self) -> str:
        return f"""
# Session Memory

## Current Task
{self.current_task}

## Key Files
{chr(10).join(f"- {x}" for x in self.key_files)}

## Errors & Fixes
{chr(10).join(f"- {x}" for x in self.errors_and_fixes)}

## Key Results
{chr(10).join(f"- {x}" for x in self.key_results)}

## Next Step
{self.next_step}
""".strip()
